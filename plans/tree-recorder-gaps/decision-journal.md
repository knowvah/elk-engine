```bash
#!/usr/bin/env bash
# Scratch mutation harness (tree-recorder-gaps D8). Never committed.
# usage: mutate.sh <file> <line> <from> <to> <test files...>
#   env MUTATE_T='<pattern>' adds `-t <pattern>` to every vitest run (mrtree: select one ref's tests).
# Applies one literal token change on one line, runs each test file under a 300 s kill timer, restores the file on
# every exit path, and prints `killed` or `survived` per test file.
set -u
if [ "$#" -lt 5 ]; then echo "usage: mutate.sh <file> <line> <from> <to> <test files...>" >&2; exit 2; fi
file=$1; line=$2; from=$3; to=$4; shift 4
root=$(git rev-parse --show-toplevel) || exit 2
cd "$root" || exit 2
[ -f "$file" ] || { echo "mutate: no such file $file" >&2; exit 2; }
git diff --quiet -- "$file" || { echo "mutate: $file is already modified; refusing" >&2; exit 2; }
lock="${TMPDIR:-/tmp}/tree-recorder-gaps/mutate.lock.$(echo "$file" | cut -d/ -f1-3 | tr '/' '_')"
if ! mkdir "$lock" 2>/dev/null; then echo "mutate: another mutant is active in this bundle ($lock)" >&2; exit 2; fi
backup=$(mktemp "${TMPDIR:-/tmp}/tree-recorder-gaps/mutate.backup.XXXXXX")
cp -p "$file" "$backup"
restore() { cp -p "$backup" "$file"; rm -f "$backup"; rmdir "$lock" 2>/dev/null; }
trap restore EXIT
trap 'exit 130' INT TERM HUP
count=$(FROM="$from" LINE="$line" perl -ne 'if ($. == $ENV{LINE}) { my $n = () = /\Q$ENV{FROM}\E/g; print $n }' "$file")
if [ "${count:-0}" != "1" ]; then
  echo "mutate: '$from' occurs ${count:-0} times on $file:$line (need exactly 1)" >&2; exit 2
fi
FROM="$from" TO="$to" LINE="$line" perl -i -pe 'if ($. == $ENV{LINE}) { s/\Q$ENV{FROM}\E/$ENV{TO}/ }' "$file"
echo "mutant: $file:$line  '$from' -> '$to'"
git --no-pager diff --no-color -U0 -- "$file" | tail -n +5
status=0
for t in "$@"; do
  log=$(mktemp "${TMPDIR:-/tmp}/tree-recorder-gaps/mutate.log.XXXXXX")
  if [ -n "${MUTATE_T:-}" ]; then
    timeout 300 npx vitest run "$t" -t "$MUTATE_T" >"$log" 2>&1; rc=$?
  else
    timeout 300 npx vitest run "$t" >"$log" 2>&1; rc=$?
  fi
  summary=$(grep -E '^ +Tests ' "$log" | tail -1 | sed 's/\x1b\[[0-9;]*m//g')
  if [ "$rc" -eq 0 ]; then echo "survived  $t  ($summary)"
  elif [ "$rc" -eq 124 ]; then echo "killed    $t  (timeout 300 s)"; status=1
  else echo "killed    $t  (exit $rc; $summary; log $log)"; status=1
  fi
done
restore; trap - EXIT
if git diff --quiet -- "$file"; then echo "restored: $file clean"; else echo "mutate: $file NOT clean after restore" >&2; exit 3; fi
exit $status
```
